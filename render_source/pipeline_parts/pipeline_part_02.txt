OUTPUTS[$scene]}"
  cp "$video" "$dest"

  echo "===== ${OUTPUTS[$scene]} =====" | tee -a diagnostics/ffprobe_all.txt
  ffprobe -v error \

    -show_entries format=duration,size \
    -of default=noprint_wrappers=1 \

  test "$(ffprobe -v error -select_streams v:0 -show_entries stream=codec_name -of csv=p=0 "$dest")" = "h264"

  test "$(ffprobe -v error -select_streams v:0 -show_entries stream=r_frame_rate -of csv=p=0 "$dest")" = "30/1"
  test "$(ffprobe -v error -select_streams v:0 -show_entries stream=pix_fmt -of csv=p=0 "$dest")" = "yuv420p"
  ffmpeg -v error -i "$dest" -f null -

  duration="$(ffprobe -v error -show_entries format=duration -of csv=p=0 "$dest")"
  midpoint="$(awk -v d="$duration" 'BEGIN{printf "%.3f", d/2}')"
  ffmpeg -y -v error -ss 00:00:01 -i "$dest" -frames:v 1 "qa/${scene}_start.png"
  ffmpeg -y -v error -ss "$midpoint" -i "$dest" -frames:v 1 "qa/${scene}_middle.png"
  ffmpeg -y -v error -sseof -2 -i "$dest" -frames:v 1 "qa/${scene}_final.png"
  test -s "qa/${scene}_start.png"
  test -s "qa/${scene}_middle.png"
  test -s "qa/${scene}_final.png"

  printf "file '%s'\n" "$(realpath "$dest")" >> diagnostics/concat.txt
done
