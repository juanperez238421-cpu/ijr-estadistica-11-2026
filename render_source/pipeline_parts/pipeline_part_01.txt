render_one() {
  local flag="$1"       # pql or pqh
  local scale="$2"
  local media_dir="$3"
  local scene="$4"
  local log_file="$5"
  local preview="0"
  if [[ "$flag" == "pql" ]]; then preview="1"; fi

  echo "===== START ${flag} ${scene} =====" | tee "$log_file"
  set +e
  docker run --rm \
    --user "$(id -u):$(id -g)" \
    -e HOME=/tmp/manim-home \
    -e STATS_TIME_SCALE="$scale" \
    -e STATS_PREVIEW="$preview" \
    -v "$PWD:/manim" \
    -w /manim \
    --entrypoint bash \
    "$IMAGE" \
    -c "
      set -euo pipefail
      mkdir -p /tmp/manim-bin
      printf '#!/usr/bin/env bash\nexit 0\n' > /tmp/manim-bin/xdg-open
      chmod +x /tmp/manim-bin/xdg-open
      export PATH=\"/tmp/manim-bin:\$PATH\"
      manim -${flag} '$SOURCE' '$scene' \
        --format=mp4 \
        --renderer=cairo \
        --media_dir='$media_dir' \
        --disable_caching
    " 2>&1 | tee -a "$log_file"
  local status=${PIPESTATUS[0]}
  set -e
  echo "===== END ${flag} ${scene} status=${status} =====" | tee -a "$log_file"
  if [[ $status -ne 0 ]]; then
    printf '%s\n' "$scene" > diagnostics/failed_scene.txt
    printf '%s\n' "$flag" > diagnostics/failed_stage.txt
    cp "$log_file" diagnostics/failure.log
    return "$status"
  fi
}

  ffmpeg -v error -i "$video" -f null -
  ffmpeg -y -v error -ss 00:00:01 -i "$video" -frames:v 1 "qa/pql_${scene}_start.png"
done
